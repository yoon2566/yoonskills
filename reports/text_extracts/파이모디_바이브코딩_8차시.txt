버튼 + LED 기본 코드
import modi_plus, time
bundle = modi_plus.MODIPlus()
button = bundle.buttons[0]
led = bundle.leds[0]
while True:
if button.clicked:
led.rgb = (255, 100, 0)
time.sleep(0.5)
led.turn_off()
elif button.pressed:
led.rgb = (0, 0, 255)
else:
led.turn_off()
time.sleep(0.1)
• 클릭 → 주황 점멸 0.5초 • 누르고 있으면 → 파랑 유지 • 아무것도 안 하면 → 꺼짐
• time.sleep(0.1): 루프 안정화 (필수!)

실습 A: 페이퍼 무드등
import modi_plus, time
bundle = modi_plus.MODIPlus()
button = bundle.buttons[0]
led = bundle.leds[0]
colors = [
(0, 0, 0), # 꺼짐
(255, 0, 0), # 빨강
(0, 255, 0), # 초록
(0, 0, 255), # 파랑
]
mode = 0
while True:
if button.double_clicked:
led.turn_off()
break
if button.clicked:
mode = (mode + 1) % 4
led.rgb = colors[mode]
time.sleep(0.1)

실습 B: 습도 알리미
import modi_plus, time
bundle = modi_plus.MODIPlus()
env = bundle.envs[0]
led = bundle.leds[0]
display = bundle.displays[0]
while True:
humid = env.humidity # 습도 값 읽기 (0~100%)
if humid > 60.0:
display.text = "WET! & Ventilate" # 디스플레이 경고
led.rgb = (0, 0, 255)
else:
display.text = "DRY & Good" # 디스플레이 정상 상태
led.rgb = (0, 255, 0)
time.sleep(1)
