실습 — 카메라에서손 감지하기
MediaPipe Hands로 카메라 영상에서 손 관절을 실시간 표시합니다
C odeW ithCathy AI 바이브코딩 × 파이모디 플러스 02
실습 A: 손 관절 실시간 표시 import cv2
import mediapipe as mp
# MediaPipe Hands 초기화
mp_hands = mp.solutions.hands
hands = mp_hands.Hands(max_num_hands=1)
mp_draw = mp.solutions.drawing_utils
cap = cv2.VideoCapture(1, cv2.CAP_DSHOW) # 카메라 설정 (노트북/기기 환경에 따라 0, 1, 2 중 선택)
while cap.isOpened():
ret, frame = cap.read()
if not ret:
print("카메라 프레임을 읽을 수 없습니다.")
break
rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) # MediaPipe 처리를 위해 BGR을 RGB로 변환
result = hands.process(rgb)
# 손이 감지되었을 때만 랜드마크 그리기
if result.multi_hand_landmarks:
for lm in result.multi_hand_landmarks:
mp_draw.draw_landmarks(frame, lm, mp_hands.HAND_CONNECTIONS)
# [수정] 손 감지 여부와 상관없이 항상 실시간 화면을 보여줍니다.
cv2.imshow("Hands (Rear Camera)", frame)
# ESC 키를 누르면 루프 탈출
if cv2.waitKey(1) == 27:
break
# [수정] while 루프가 완전히 끝난 뒤(정상 종료 시)에 카메라와 창을 닫습니다.
cap.release()
cv2.destroyAllWindows()

손가락 관절 좌표의 y값을 비교해 손 들기, 주먹, V 제스처를 판별합니다

손가락 펴짐 판별 코드
# 손가락이 펴져있는지 확인
# tip(끝) y < pip(관절) y → 펴짐
tips = [8, 12, 16, 20] # 검지~새끼
fingers_up = []
for tip in tips:
if lm.landmark[tip].y < lm.landmark[tip-2].y:
fingers_up.append(True)
else:
fingers_up.append(False)
count = sum(fingers_up) # 펴진 손가락 수
핵심: 손가락 끝(tip)의 y값이 관절(pip)의 y값보다 작으면(위쪽
쪽이면) 펴져있다고 판단합니다!

import cv2
import mediapipe as mp
import modi_plus
bundle = modi_plus.MODIPlus()
led = bundle.leds[0]
display = bundle.displays[0]
mp_hands = mp.solutions.hands
hands = mp_hands.Hands(max_num_hands=1)
mp_draw = mp.solutions.drawing_utils
cap = cv2.VideoCapture(0)
colors = {
0:(0,0,0), 1:(255,0,0),
2:(0,0,255), 3:(0,255,0),
4:(255,255,0), 5:(255,255,255)
}
while cap.isOpened():
ret, frame = cap.read()
if not ret: break
rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
result = hands.process(rgb)
count = 0
if result.multi_hand_landmarks:
lm = result.multi_hand_landmarks[0]
mp_draw.draw_landmarks(
frame, lm, mp_hands.HAND_CONNECTIONS)
tips = [8, 12, 16, 20]
for tip in tips:
if lm.landmark[tip].y < lm.landmark[tip-2].y:
count += 1
led.rgb = colors.get(count, (0,0,0))
display.text = f"F:{count}"
cv2.imshow("Gesture", frame)
if cv2.waitKey(1) == 27: break
