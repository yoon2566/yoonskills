음성 인식 + 명령어 판별 + 파이모디 제어 전체 코드입니다
C odeWithCathy AI 바이브코딩 × 파이모디 플러스 03
import speech_recognition as sr
import modi_plus, time
# 파이모디 연결
bundle = modi_plus.MODIPlus()
led = bundle.leds[0]
display = bundle.displays[0]
# 음성 인식기
recognizer = sr.Recognizer()
print("
 음성 파이모디 시작!")
while True:
with sr.Microphone() as source:
print("\n말해보세요...")
recognizer.adjust_for_ambient_noise(
source)
audio = recognizer.listen(source)
try:
text = recognizer.recognize_google(
audio, language="ko-KR")
print(f"인식: {text}")
if "켜" in text:
led.rgb = (255,255,255)
display.text = "LED ON"
elif "꺼" in text:
led.turn_off()
display.text = "LED OFF"
elif "빨간" in text:
led.rgb = (255,0,0)
display.text = "RED"
elif "파란" in text:
led.rgb = (0,0,255)
display.text = "BLUE"
elif "종료" in text:
led.turn_off()
display.text = "BYE"
break
except:
print("다시 말해주세요")

음성 + 센서 연동 매핑표
음성 명령 키워드 센서 파이모디 동작
"온도 알려줘" 온도 env.tem p erature D isp lay와 터미널창에 온도 표시 + 30 도 초과시 LED 빨강
"습도 알려줘" 습도 env.hum id ity D isp lay와 터미널창에 습도 표시 + 70% 초과시 LED 파랑
"밝기 알려줘" 밝기 env.intensity D isp lay와 터미널창에 밝기 % 표시 + 30% 이하면 LED 노랑
"전체 상태" 상태 모든 센서 온도·습도·밝기를 2초 간격으로 순서대로 D isp lay와 터미널창에 표시
센서 연동 코드 예시
elif "온도" in text:
t = env.temperature
display.text = f"T:{t:.1f}C"
if t > 30:
led.rgb = (255,0,0)
else:
led.rgb = (0,255,0)
print(f"온도: {t:.1f}C")
elif "상태" in text:
for label, val in [
("T", env.temperature),
("H", env.humidity),
("L", env.illuminance)]:
display.text = f"{label}:{val:.0f}"
time.sleep(2)
