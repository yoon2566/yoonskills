실습 A: 날씨→LED 기본
import requests, modi_plus, time
API_KEY = "YOUR_KEY"
bundle = modi_plus.MODIPlus()
led = bundle.leds[0]
display = bundle.displays[0]
url = (f"http://api.openweathermap.org"
f"/data/2.5/weather?q=Seoul"
f"&appid={API_KEY}&units=metric&lang=kr")
data = requests.get(url).json()
w = data["weather"][0]["main"]
t = data["main"]["temp"]
if w == "Clear":
led.rgb = (255,255,0)
display.text = f"SUNNY {t:.0f}C"
elif w in ["Rain","Drizzle"]:
led.rgb = (0,0,255)
display.text = f"RAIN {t:.0f}C"
else:
led.rgb = (150,150,150)
display.text = f"{w} {t:.0f}C"
실습 B: 자동 갱신 (5분마다)
# 5분마다 날씨 자동 갱신
while True:
data = requests.get(url).json()
w = data["weather"][0]["main"]
t = data["main"]["temp"]
h = data["main"]["humidity"]
d = data["weather"][0]["description"]
# LED 색상 설정
colors = {
"Clear":(255,255,0),
"Clouds":(150,150,150),
"Rain":(0,0,255),
"Snow":(255,255,255),
}
led.rgb = colors.get(w,(100,100,100))
display.text = f"{d} {t:.0f}C"
print(f"{d} {t}C {h}%")
time.sleep(300) # 5분

전체 코드 — 날씨 스테이션 완성
날씨 API + LED + Display + 자동 갱신 전체 통합 코드입니다

import requests, modi_plus, time
API_KEY = "YOUR_API_KEY"
CITY = "Seoul"
bundle = modi_plus.MODIPlus()
led = bundle.leds[0]
display = bundle.displays[0]
def blink(color, n=3):
for _ in range(n):
led.rgb = color
time.sleep(0.3)
led.turn_off()
time.sleep(0.3)
def get_weather():
url = (f"http://api.openweathermap.org"
f"/data/2.5/weather?q={CITY}"
f"&appid={API_KEY}"
f"&units=metric&lang=kr")
return requests.get(url).json()
def set_modi(data):
w = data["weather"][0]["main"]
t = data["main"]["temp"]
d = data["weather"][0]["description"]
colors = {
"Clear":(255,255,0),
"Clouds":(150,150,150),
"Rain":(0,0,255),
"Drizzle":(0,100,255),
"Snow":(255,255,255),
"Thunderstorm":(255,0,0),
}
if w == "Thunderstorm":
blink((255,0,0))
else:
led.rgb = colors.get(w,(100,100,100))
display.text = f"{d} {t:.0f}C"
print("Weather Station Start!")
while True:
data = get_weather()
set_modi(data)
time.sleep(300)
