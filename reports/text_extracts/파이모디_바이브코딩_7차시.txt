색상 R G B 코드
빨강 255 0 0 led .rg b = (255, 0, 0)
초록 0 255 0 led.rg b = (0, 255, 0)

꺼짐 0 0 0 led.turn_off()

import modi_plus, time
bundle = modi_plus.MODIPlus()
led = bundle.leds[0]
display = bundle.displays[0]
messages = [
("Good Morning!", (255, 200, 50)),
("Study Time!", (0, 150, 255)),
("Break Time!", (0, 255, 100)),
("Good Night!", (100, 0, 200)),
]
for msg, color in messages:
display.text = msg
led.rgb = color
time.sleep(3)
display.text = "Bye!"
led.turn_off()
