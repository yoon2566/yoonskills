함수 def — 코드를 묶어서 이름 붙이기
같은 코드를 반복하지 않고, 함수로 한 번 만들어 여러 번 재사용합니다
C odeWithCathy AI 바이브코딩 × 파이모디 플러스 01
함수가 왜 필요한가?
함수의 3가지 장점:
① 반복 코드 제거 — 같은 코드를 여러 번 쓸 필요 없음
② 재사용 가능 — 한 번 만들면 어디서든 호출
③ 유지보수 쉬움 — 수정이 필요하면 함수 하나만 고침
def 구조
# 함수 정의
def greet(name, age):
"""인사 함수""" # 독스트링(설명)
msg = f"{name}({age}세) 안녕!"
return msg # 결과 반환
# 함수 호출
result = greet("Alice", 17)
print(result) # Alice(17세) 안녕!
# return 없는 함수 (출력만)
def say_hi(name):
print(f"안녕, {name}!")
say_hi("Bob") # 안녕, Bob!
return 유무 비교
# return 있는 함수 (값 돌려줌)
def add(a, b):
return a + b
result = add(3, 5) # 8
※ return이 있으면 결과를 변수에 저장 가능, 없으면 None 반환 (출력 전용 함수)

실습 A: 온도 변환기
# 실습 A: 온도 변환 함수
def c_to_f(c):
"""섭씨 → 화씨"""
return round(c * 9/5 + 32, 1)
def f_to_c(f):
"""화씨 → 섭씨"""
return round((f - 32) * 5/9, 1)
temp = float(input("온도: "))
unit = input("단위(C/F): ").upper()
if unit == "C":
print(f"{temp}C = {c_to_f(temp)}F")
else:
print(f"{temp}F = {f_to_c(temp)}C")
실습 B: 기본값 매개변수
# 실습 B: 기본값 매개변수
def greet(name, msg="안녕!"):
print(f"{name}: {msg}")
greet("Alice") # Alice: 안녕!
greet("Bob", "반가워!") # Bob: 반가워!
# 실습: 계산기 함수
def calc(a, b, op="+"):
if op == "+": return a + b
elif op == "-": return a - b
elif op == "*": return a * b
elif op == "/": return a / b
print(calc(10, 3)) # 13
print(calc(10, 3, "*")) # 30

# 실습 A: 가위바위보
import random
choices = ["가위", "바위", "보"]
wins = 0
for _ in range(3): # 3판
user = input("가위/바위/보: ")
com = random.choice(choices)
print(f"컴퓨터: {com}")
if user == com:
print("비김!")
elif (user=="가위" and com=="보") or \
(user=="바위" and com=="가위") or \
(user=="보" and com=="바위"):
print("이겼다!")
wins += 1
else:
print("졌다!")
print(f"\n3판 중 {wins}승!")
실습 B: 반응 속도 테스트
# 실습 B: 반응 속도 테스트
import time, random
print("준비...")
time.sleep(random.uniform(2, 5))
print("지금! Enter를 눌러!")
start = time.time()
input()
end = time.time()
reaction = (end - start) * 1000
print(f"반응속도: {reaction:.0f}ms")
if reaction < 300:
print("번개 반사신경!")
elif reaction < 500:
print("꽤 빠르네요!")
else:
print("좀 더 연습!")

전체 코드
todos = []
def show():
if not todos: print("할 일이 없습니다")
for i, t in enumerate(todos, 1):
print(f" {i}. {t}")
def add():
t = input("추가할 일: ")
todos.append(t)
print(f"추가됨: {t}")
def delete():
show()
n = int(input("삭제 번호: ")) - 1
if 0 <= n < len(todos):
print(f"삭제됨: {todos.pop(n)}")
# 메인 루프
while True:
print("\n=== 할 일 관리 ===")
print("1.보기 2.추가 3.삭제 4.종료")
choice = input("선택: ")
if choice == "1":
show()
elif choice == "2":
add()
elif choice == "3":
delete()
elif choice == "4":
print("종료합니다!")
break
else:
print("잘못된 선택")
• 기능별 함수 분리 (show/add/delete) → 유지보수 쉬움
• while True + break → 메뉴 루프 • enumerate() → 번호 자동 부여
